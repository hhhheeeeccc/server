package servicedesk

// Plugin RPC registers plain maps and []any on both peers. A private Go struct
// inside Post.Props cannot cross that boundary and can close the RPC stream.
func serviceNotificationRPCProps(card serviceNotificationCard) map[string]any {
	props := map[string]any{
		"version": card.Version, "key": card.Key, "event": card.Event,
		"title": card.Title, "service": card.Service,
		"status": card.Status, "status_label": card.StatusLabel,
		"priority": card.Priority, "captured_at": card.CapturedAt,
		"reporter": notificationRPCPerson(card.Reporter),
		"assignee": notificationRPCPerson(card.Assignee),
	}
	if card.Description != "" {
		props["description"] = card.Description
	}
	if card.URL != "" {
		props["url"] = card.URL
	}
	if card.ApprovalDue != 0 {
		props["approval_due_at"] = card.ApprovalDue
	}
	if len(card.Fields) != 0 {
		fields := make([]any, 0, len(card.Fields))
		for _, field := range card.Fields {
			fields = append(fields, map[string]any{"label": field.Label, "value": field.Value})
		}
		props["fields"] = fields
	}
	return props
}

func notificationRPCPerson(person notificationPerson) map[string]any {
	props := map[string]any{"name": person.Name}
	if person.ID != "" {
		props["id"] = person.ID
	}
	return props
}
